cmake_minimum_required(VERSION 3.24)
project(Musix VERSION 0.12.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT APPLE)
    message(FATAL_ERROR "Musix supports Linux and macOS only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo)
if(ipo)
 set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
# macOS has no session bus, so MPRIS and the freedesktop notifier are built
# only where they can work. Everything that follows keys off this one answer.
if(APPLE)
 set(SUNG_DBUS OFF)
else()
 set(SUNG_DBUS ON)
endif()
# Directory scope, not per-target: the tests compile playbacknotifier.cpp too,
# and a stubbed notifier under a test that asserts it posts would pass silently.
if(SUNG_DBUS)
 add_compile_definitions(SUNG_DBUS)
endif()
set(SUNG_QT_COMPONENTS Core Gui Quick Qml QuickControls2 Multimedia Network Svg)
if(SUNG_DBUS)
 list(APPEND SUNG_QT_COMPONENTS DBus)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS ${SUNG_QT_COMPONENTS})
qt_policy(SET QTP0004 NEW)
# The backend every executable shares. mpris.cpp is deliberately absent: only
# the app exposes MPRIS, and only where there is a bus to expose it on.
set(SUNG_BACKEND_SOURCES src/scrobbler.cpp src/scrobbler.h src/m3color.cpp src/m3color.h src/m3shape.cpp src/m3shape.h src/musicserver.cpp src/musicserver.h src/jellyfin.cpp src/jellyfin.h src/backendserver.cpp src/subsonic.cpp src/subsonic.h src/backend.cpp src/productfeatures.cpp src/presentationfeatures.cpp src/backend.h src/collectionview.h src/rowselection.h src/playbacknotifier.cpp src/playbacknotifier.h)
set(SUNG_BACKEND_LIBS Qt6::Core Qt6::Gui Qt6::Multimedia Qt6::Network)
if(SUNG_DBUS)
 list(APPEND SUNG_BACKEND_LIBS Qt6::DBus)
endif()
# Keep the binary at build/musix rather than build/musix.app so the scripts and
# the diagnostics harness keep addressing it the same way on either platform.
set(CMAKE_MACOSX_BUNDLE FALSE)
qt_add_executable(musix src/windowresources.h src/artworkurl.h src/desktoptheme.cpp src/desktoptheme.h src/main.cpp src/roundedart.cpp src/roundedart.h src/motionartwork.cpp src/motionartwork.h ${SUNG_BACKEND_SOURCES})
target_link_libraries(musix PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Multimedia Qt6::Network Qt6::Svg)
if(SUNG_DBUS)
 target_sources(musix PRIVATE src/mpris.cpp src/mpris.h)
 target_link_libraries(musix PRIVATE Qt6::DBus)
endif()
target_compile_options(musix PRIVATE -Wall -Wextra -Wpedantic "$<$<CONFIG:Release>:-Oz;-ffunction-sections;-fdata-sections>")
if(APPLE)
 target_link_options(musix PRIVATE "$<$<CONFIG:Release>:-Oz;-Wl,-dead_strip>")
else()
 target_link_options(musix PRIVATE "$<$<CONFIG:Release>:-Oz;-Wl,--gc-sections;-Wl,-z,pack-relative-relocs>")
endif()
set_target_properties(musix PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
# This small, first-use-only component can compile on demand. Keep the player
# and animation components precompiled, without embedding unused picker bytecode.
set_source_files_properties(qml/FileDialogs.qml PROPERTIES QT_QML_SKIP_CACHEGEN TRUE)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set_target_properties(musix PROPERTIES QT_QMLCACHEGEN_ARGUMENTS "--only-bytecode")
qt_add_qml_module(musix URI SungUi VERSION 1.0 NO_PLUGIN NO_RESOURCE_TARGET_PATH RESOURCE_PREFIX "/"
 QML_FILES qml/MDockedToolbar.qml qml/MCheckbox.qml qml/MRadioButton.qml qml/MExposedDropdown.qml qml/MCard.qml qml/MNavigationDrawer.qml qml/MBottomSheet.qml qml/MShape.qml qml/MRichTooltip.qml qml/MTooltip.qml qml/MDragHandle.qml qml/MWavyProgress.qml qml/MAppBarRow.qml qml/MElevation.qml qml/MBadge.qml qml/MNavigationBar.qml qml/MPullToRefresh.qml qml/MCarousel.qml qml/MFloatingToolbar.qml qml/MFabMenu.qml qml/MSplitButton.qml qml/PlaylistVersions.qml qml/ListeningStats.qml qml/SingAlong.qml qml/ArtistHero.qml qml/NavigationTransition.qml qml/MSettingRow.qml qml/Onboarding.qml qml/AmbientBackdrop.qml qml/ImmersiveCoverflow.qml qml/AccentPicker.qml qml/PlaybackHud.qml qml/VolumeControl.qml qml/ViewLayoutDialog.qml qml/HomeEditor.qml qml/OutputPicker.qml qml/ListeningSessions.qml qml/PlaylistCoverDialog.qml qml/CommandPalette.qml qml/ArtworkControls.qml qml/MatchText.qml qml/PlayingIndicator.qml qml/TrackPresentation.qml qml/PlaylistCover.qml qml/CatalogSkeleton.qml qml/SmartPlaylistDialog.qml qml/TrackDetailsDialog.qml qml/ShortcutHelp.qml qml/MScrollBar.qml qml/MTextField.qml qml/MSearchField.qml qml/MSegmentedControl.qml qml/ServerConnection.qml qml/ServerToolbar.qml qml/MChip.qml qml/LibraryTabs.qml qml/MLoadingIndicator.qml qml/TrackList.qml qml/SelectionBar.qml qml/FileDialogs.qml qml/SettingSlider.qml qml/LyricsView.qml qml/ImmersivePlayer.qml qml/MiniPlayer.qml qml/MMenu.qml qml/MMenuItem.qml qml/MDivider.qml qml/MDialog.qml qml/MSwitch.qml qml/Main.qml qml/Theme.qml qml/Icon.qml qml/MButton.qml qml/MNavigationItem.qml qml/SungText.qml qml/ArtworkViewer.qml qml/PlaybackGlyph.qml qml/Artwork.qml qml/TrackRow.qml qml/ArtCard.qml qml/SeekBar.qml)
file(GLOB icons CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" assets/icons/*.svg)
qt_add_resources(musix assets PREFIX "/" FILES ${icons}
 OPTIONS --compress 19 --threshold 0)
if(CMAKE_BUILD_TYPE STREQUAL "Release" AND NOT APPLE)
 add_custom_command(TARGET musix POST_BUILD COMMAND ${CMAKE_STRIP} --strip-unneeded $<TARGET_FILE:musix>)
endif()
include(GNUInstallDirs)
install(TARGETS musix RUNTIME DESTINATION bin)
install(FILES helper/catalog.py helper/online_artwork.py helper/requirements.txt DESTINATION lib/musix)
if(NOT APPLE)
 install(FILES assets/musix.desktop DESTINATION share/applications)
 install(FILES assets/sung.png DESTINATION share/icons/hicolor/512x512/apps RENAME musix.png)
endif()
install(FILES LICENSE NOTICE licenses/MaterialSymbols-LICENSE.txt DESTINATION share/licenses/musix)
include(CTest)
if(BUILD_TESTING)

 qt_add_executable(musix-crossfade-tests tests/crossfade_test.cpp ${SUNG_BACKEND_SOURCES})

 target_link_libraries(musix-crossfade-tests PRIVATE ${SUNG_BACKEND_LIBS} Qt6::Test)
 add_test(NAME crossfade COMMAND musix-crossfade-tests)
 set_tests_properties(crossfade PROPERTIES TIMEOUT 400 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SUNG_FIXTURE_HELPER=${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog_fixture.py;QT_FFMPEG_DECODING_HW_DEVICE_TYPES=,;QT_FFMPEG_ENCODING_HW_DEVICE_TYPES=,")
 qt_add_executable(musix-m3color-tests tests/m3color_test.cpp src/m3color.cpp src/m3color.h src/m3shape.cpp src/m3shape.h)
 target_include_directories(musix-m3color-tests PRIVATE src)
 target_link_libraries(musix-m3color-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
 add_test(NAME m3color COMMAND musix-m3color-tests)
 qt_add_executable(musix-subsonic-protocol-tests tests/subsonic_protocol_test.cpp src/subsonic.cpp src/subsonic.h)
 target_include_directories(musix-subsonic-protocol-tests PRIVATE src)
 target_link_libraries(musix-subsonic-protocol-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
 add_test(NAME subsonic-protocol COMMAND musix-subsonic-protocol-tests)
 qt_add_executable(musix-subsonic-tests tests/subsonic_test.cpp ${SUNG_BACKEND_SOURCES})
 target_include_directories(musix-subsonic-tests PRIVATE src)
 target_link_libraries(musix-subsonic-tests PRIVATE ${SUNG_BACKEND_LIBS} Qt6::Test)

 qt_add_executable(musix-jellyfin-protocol-tests tests/jellyfin_protocol_test.cpp src/jellyfin.cpp src/jellyfin.h src/subsonic.cpp src/subsonic.h)
 target_include_directories(musix-jellyfin-protocol-tests PRIVATE src)
 target_link_libraries(musix-jellyfin-protocol-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Test)
 add_test(NAME jellyfin-protocol COMMAND musix-jellyfin-protocol-tests)
 qt_add_executable(musix-jellyfin-tests tests/jellyfin_test.cpp ${SUNG_BACKEND_SOURCES})
 target_include_directories(musix-jellyfin-tests PRIVATE src)
 target_link_libraries(musix-jellyfin-tests PRIVATE ${SUNG_BACKEND_LIBS} Qt6::Test)

 qt_add_executable(musix-artwork-tests tests/artwork_test.cpp src/artworkurl.h src/m3shape.cpp src/m3shape.h src/roundedart.cpp src/roundedart.h src/motionartwork.cpp src/motionartwork.h)
 target_include_directories(musix-artwork-tests PRIVATE src)
 target_link_libraries(musix-artwork-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Network Qt6::Multimedia Qt6::Test)
 add_test(NAME artwork COMMAND musix-artwork-tests)
 set_tests_properties(artwork PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
 qt_add_executable(musix-tests tests/backend_test.cpp src/artworkurl.h src/desktoptheme.cpp src/desktoptheme.h ${SUNG_BACKEND_SOURCES})
 target_include_directories(musix-tests PRIVATE src)
 target_link_libraries(musix-tests PRIVATE ${SUNG_BACKEND_LIBS} Qt6::Test)
 add_test(NAME backend COMMAND musix-tests)
 set_tests_properties(backend PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SUNG_FIXTURE_HELPER=${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog_fixture.py;QT_FFMPEG_DECODING_HW_DEVICE_TYPES=,;QT_FFMPEG_ENCODING_HW_DEVICE_TYPES=,")
 # The notifier speaks to org.freedesktop.Notifications, so both the test and
 # the bus it needs exist only where there is a session bus.
 if(SUNG_DBUS)
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  qt_add_executable(musix-notification-tests tests/notification_test.cpp ${SUNG_BACKEND_SOURCES})
  target_include_directories(musix-notification-tests PRIVATE src)
  target_link_libraries(musix-notification-tests PRIVATE ${SUNG_BACKEND_LIBS} Qt6::Test)
  add_test(NAME notifications COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:musix-notification-tests>)
  set_tests_properties(notifications PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;SUNG_FIXTURE_HELPER=${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog_fixture.py;QT_FFMPEG_DECODING_HW_DEVICE_TYPES=,;QT_FFMPEG_ENCODING_HW_DEVICE_TYPES=,")
 endif()
endif()

 find_package(Qt6 REQUIRED COMPONENTS Test GuiPrivate)
